curl https://models.bytefuture.ai/v1/chat/completions \
  -H "Authorization: Bearer gw-•••" \
  -H "Content-Type: application/json" \
  -H "X-Hint-Step: execute" \
  -d '{
  "model": "auto",
  "messages": [
    {
      "role": "user",
      "content": "用一句话解释什么是向量嵌入？"
    },
    {
      "role": "assistant",
      "content": "向量嵌入是把文本、图像等内容映射成一串数字（高维向量）的技术，语义相近的内容在向量空间中彼此靠近，从而支持相似度搜索、聚类和推荐等任务。"
    },
    {
      "role": "user",
      "content": "帮我写一个通过这个网关调用聊天接口的 Python 示例。"
    }
  ],
  "provider": {
    "order": ["glm", "deepseek"],
    "max_price": { "completion": 3.0 },
    "zdr": true
  },
  "stream": true
}'
